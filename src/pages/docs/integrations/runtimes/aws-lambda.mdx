## About

Wrap your Lambda handler to see one invocation span and the work it called. For example, when Ridgeline Outfitters answers a stock question, you can see whether the function started cold, whether the stock lookup ran, and whether the handler returned or failed. The wrapper attempts to send completed spans before Lambda freezes the execution environment.

## 1. Install the wrapper for your runtime

Install the dependencies in your function project and include them in your deployment package. The Python SDK supports Python 3.10 through 3.15. Choose a Python runtime available for your function. For TypeScript, use a promise handler on Node.js 22, 24, or 26. For Go, start in a new module directory and build with Go 1.26 or later. If `go` is not already on your `PATH`, replace `/path/to/go` with your Go installation directory.

## 2. Wrap the handler once

Save the file for your runtime. Build the provider and wrap the handler **outside** the invocation so warm calls can reuse the provider. If you already have an OpenTelemetry provider with a batch processor and exporter, pass it to the wrapper instead of creating another. Remove any hand-written invocation span and flush, or overlapping Lambda auto-instrumentation, including an auto-instrumenting layer. Keep model and tool instrumentors that trace work *inside* the handler.

<CodeGroup titles={["Python: stock.py", "TypeScript: index.ts", "Go: main.go"]}>

```python
import os

from fi_instrumentation import register
from fi_instrumentation.aws_lambda import LambdaTracer
from fi_instrumentation.fi_types import ProjectType

provider = register(project_name=os.environ["FI_PROJECT_NAME"],
                    project_type=ProjectType.OBSERVE, batch=True,
                    timeout=2, verbose=False)
tracer = provider.get_tracer("ridgeline.stock")
lifecycle = LambdaTracer(provider)
stock = {"TRAIL-40": 6}

@lifecycle.wrap
def handler(event, context):
    sku = (event.get("queryStringParameters") or {}).get("sku")
    with tracer.start_as_current_span("lookup_stock", attributes={"fi.span.kind": "TOOL"}) as lookup:
        trace_id = f"{lookup.get_span_context().trace_id:032x}"
        quantity = stock.get(sku)
    if quantity is None:
        raise LookupError("Stock snapshot unavailable")
    return {"statusCode": 200, "headers": {"x-trace-id": trace_id}, "body": f"{quantity} packs available"}
```

```typescript
import { ProjectType, register, traceLambdaHandler } from "@traceai/fi-core";

const provider = register({
  projectName: process.env.FI_PROJECT_NAME, projectType: ProjectType.OBSERVE, batch: true, verbose: false, setGlobalTracerProvider: true,
});
const tracer = provider.getTracer("ridgeline.stock");
const stock = new Map([["TRAIL-40", 6]]);
type StockEvent = { queryStringParameters?: { sku?: string } | null };

export const handler = traceLambdaHandler(async (event: StockEvent) => {
  const lookup = tracer.startSpan("lookup_stock", { attributes: { "fi.span.kind": "TOOL" } });
  let quantity: number | undefined;
  try {
    quantity = stock.get(event.queryStringParameters?.sku ?? "");
  } finally {
    lookup.end();
  }
  if (quantity === undefined) throw new Error("Stock snapshot unavailable");
  return { statusCode: 200, headers: { "x-trace-id": lookup.spanContext().traceId }, body: `${quantity} packs available` };
}, { provider });
```

```go
package main

import (
    "cmp"
    "context"
    "errors"
    "fmt"
    "os"
    "strings"

    "github.com/aws/aws-lambda-go/lambda"
    traceailambda "github.com/future-agi/traceAI/go/lambda"
    "go.opentelemetry.io/otel/attribute"
    "go.opentelemetry.io/otel/exporters/otlp/otlptrace/otlptracehttp"
    "go.opentelemetry.io/otel/sdk/resource"
    sdktrace "go.opentelemetry.io/otel/sdk/trace"
    "go.opentelemetry.io/otel/trace"
)

type stockEvent struct {
    Query map[string]string `json:"queryStringParameters"`
}

func wrapped() lambda.Handler {
    base := cmp.Or(os.Getenv("FI_TRACES_URL"), os.Getenv("FI_BASE_URL"), "https://api.futureagi.com")
    project := os.Getenv("FI_PROJECT_NAME")
    opts := []sdktrace.TracerProviderOption{sdktrace.WithResource(resource.NewSchemaless(
        attribute.String("project_name", project), attribute.String("project_type", "observe"),
        attribute.String("service.name", project)))}
    key, secret := os.Getenv("FI_API_KEY"), os.Getenv("FI_SECRET_KEY")
    if project != "" && key != "" && secret != "" {
        exporter, err := otlptracehttp.New(context.Background(),
            otlptracehttp.WithEndpointURL(strings.TrimRight(base, "/") + "/tracer/v1/traces"),
            otlptracehttp.WithHeaders(map[string]string{"X-Api-Key": key, "X-Secret-Key": secret}),
            otlptracehttp.WithTimeout(traceailambda.FlushTimeout),
        )
        if err == nil { opts = append(opts, sdktrace.WithBatcher(exporter)) }
    }
    provider := sdktrace.NewTracerProvider(opts...)
    tracer := provider.Tracer("ridgeline.stock")
    stock := map[string]int{"TRAIL-40": 6}
    handle := func(ctx context.Context, event stockEvent) (map[string]any, error) {
        _, lookup := tracer.Start(ctx, "lookup_stock", trace.WithAttributes(attribute.String("fi.span.kind", "TOOL")))
        quantity, ok := stock[event.Query["sku"]]
        lookup.End()
        if !ok { return nil, errors.New("stock snapshot unavailable") }
        return map[string]any{"statusCode": 200, "headers": map[string]string{"x-trace-id": lookup.SpanContext().TraceID().String()}, "body": fmt.Sprintf("%d packs available", quantity)}, nil
    }
    return traceailambda.WrapHandler(lambda.NewHandler(handle), provider)
}

func main() {
    handler := wrapped()
    if len(os.Args) == 3 && os.Args[1] == "--event" { // Invoke locally with a saved event.
        payload, err := os.ReadFile(os.Args[2])
        if err != nil { panic(err) }
        response, err := handler.Invoke(context.Background(), payload)
        if err != nil { panic(err) }
        fmt.Println(string(response))
        return
    }
    lambda.StartHandler(handler)
}
```

</CodeGroup>

For Python, set the handler to `stock.handler`. For TypeScript, compile with `npx tsc index.ts --target ES2022 --module node16 --skipLibCheck` and use `index.handler`. If you supply your own Node.js provider, install an async context manager so child spans attach across `await`; `register()` does this when registering the global provider. For Go, pass the same provider and invocation context to any model instrumentor. Run `go mod tidy`, then build `bootstrap` for your function architecture, for example `GOOS=linux GOARCH=arm64 CGO_ENABLED=0 go build -o bootstrap .` for arm64 on `provided.al2023`.

## 3. Invoke the function and inspect the trace

Save this HTTP API v2 event as `event.json`. You can send it to your deployed function or invoke the wrapped handler locally. The synthetic SKU is present in the stock snapshot, so the handler returns status `200` and the body `6 packs available`.

```json
{
  "version": "2.0",
  "requestContext": { "http": { "method": "GET" } },
  "queryStringParameters": { "sku": "TRAIL-40" }
}
```

To try the event locally without deploying to AWS, use the command for your runtime. Each successful response includes an `x-trace-id` header you can use to find that invocation.

<CodeGroup titles={["Python", "TypeScript (after compiling)", "Go (after go mod tidy)"]}>

```bash
python -c 'import json, stock; print(json.dumps(stock.handler(json.load(open("event.json")), None)))'
```

```bash
node -e 'require("./index.js").handler(require("./event.json"), { getRemainingTimeInMillis: () => 30000 }).then(r => console.log(JSON.stringify(r)))'
```

```bash
go run . --event event.json
```

</CodeGroup>

1. Under **Observe**, select **Tracing**, open `ridgeline-stock`, then select the trace. Look for `lookup_stock` directly beneath `lambda.handler` (Python and Go) or `aws.lambda.invoke` (TypeScript). Without an incoming parent, the invocation starts its own trace.
2. Send the event again. Each attempt has its own invocation span. Look for the Lambda request ID in `faas.invocation_id`. `faas.coldstart` is `true` on the first call to a wrapper and `false` if Lambda reuses that execution environment. Two consecutive calls need not reach the same environment.
3. Change the SKU to `TRAIL-99`. The handler error still reaches the runtime. The lookup completed, but the **invocation** span has `ERROR` status and `error.type`. Retry the failed event. The retry gets a new invocation span; `faas.coldstart` tells you whether the wrapper has run before, not whether this event is a retry.
4. To follow an upstream HTTP request, pass its W3C `traceparent` in `headers` on the event. The invocation becomes a child of that span in the same trace. The wrapper does not forward your Future AGI credentials to downstream services.

## Runtime differences and limits

| Behavior | Python | TypeScript | Go |
| --- | --- | --- | --- |
| Handler completion | Synchronous handlers only. Exceptions are re-raised. | The default observes the returned value or promise. A rejected promise or thrown error remains a failure. | The adapter wraps `lambda.NewHandler(handle)` and preserves returned errors and panics. Pass `ctx` into child calls. |
| Remote parent | API Gateway v1/v2, function URL, or ALB. Supports `multiValueHeaders`. | API Gateway v1/v2, function URL, or ALB. Conflicting parent headers are ignored. | API Gateway HTTP API v2 or function URL only. |
| HTTP proxy response | Integer `statusCode` from 100 to 599 becomes `http.response.status_code`. A 5xx sets `ERROR` with the code as `error.type`; a 4xx alone does not. | Same as Python for recognized HTTP proxy events. | Returning a 5xx response with a nil Go error does **not** by itself mark the invocation as failed. |

Direct invocations and queue batches start independent traces. The wrappers read the HTTP trace parent, not baggage, X-Ray headers, event bodies, or arbitrary request headers. Invocation spans do not include the function result, exception message, or stack trace. Your own spans and model instrumentors have separate content settings. The old copied helper's `LAMBDA_TRACE_CAPTURE_CONTENT` switch does not apply.

**Node.js callbacks and streaming.** On Node.js 22, choose `completion: "callback"` for a callback handler, even if its TypeScript signature looks like an async handler. Set `context.callbackWaitsForEmptyEventLoop = false` *before* entering the wrapped handler and keep it false through completion. If the flag is true or unset on entry, the original handler runs without invocation tracing or a flush. Finish required work before calling back. Use promise handlers on Node.js 24 and later. Node.js response-streaming handlers pass through without invocation tracing.

**Go function URLs.** Return JSON as a value that does not implement `io.Reader`. For a raw reader response, its JSON encoding must fail or produce an empty object. The wrapper preserves Content-Type for these supported responses, but reads the whole reader before delivery; it does not stream incrementally. Send `{}` rather than an empty payload to a handler with no event fields. Use one invocation at a time per execution environment (`AWS_LAMBDA_MAX_CONCURRENCY` unset or `1`) with `aws-lambda-go` v1.55.1.

**Timeouts and late spans.** By default, after ending the invocation span, each wrapper waits up to 2,000 ms for the batch to flush, leaving 250 ms of the invocation's remaining time for the runtime. For example, with 400 ms remaining, the maximum wait is 150 ms. An exhausted budget skips the flush; a hard timeout can lose spans. A flush failure does not replace the handler's result or error. This budget limits the wait for export, not time spent in application code or a synchronous span processor.

Python exposes `flush_timeout_millis` and `reserve_millis`; TypeScript exposes `maxFlushWaitMillis` and `remainingTimeReserveMillis`. A timed-out wait does not cancel an exporter: Python skips a new flush while the previous one is running, and TypeScript shares a pending flush for the provider. A custom Go provider must honor the flush context deadline. Use a batch processor with a finite exporter timeout and do not shut down the provider after each invocation.
